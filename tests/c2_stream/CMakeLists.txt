cmake_minimum_required(VERSION 3.24)
project(C2PauseProbe LANGUAGES CXX)
set(C2_GENERATED "" CACHE PATH "Pinned generated adapter directory")
if(NOT C2_GENERATED)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(C2_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_pause.py"
    --header "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/stb/stb_image_resize2.h"
    --output "${C2_GENERATED}" COMMAND_ERROR_IS_FATAL ANY)
endif()
foreach(mode simd scalar)
  add_executable(c2_pause_${mode} pause_probe.cpp)
  target_compile_features(c2_pause_${mode} PRIVATE cxx_std_20)
  target_include_directories(c2_pause_${mode} PRIVATE "${C2_GENERATED}")
  if(TARGET thumbgen_core)
    target_compile_definitions(c2_pause_${mode} PRIVATE C2_REAL_SOURCE)
    target_include_directories(c2_pause_${mode} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src" "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/skcms")
    target_link_libraries(c2_pause_${mode} PRIVATE thumbgen_core OpenImageIO::OpenImageIO)
  endif()
  if(mode STREQUAL scalar)
    target_compile_definitions(c2_pause_${mode} PRIVATE STBIR_NO_SIMD)
  endif()
  if(MSVC)
    target_compile_options(c2_pause_${mode} PRIVATE /O2 /fp:precise /utf-8)
  else()
    target_compile_options(c2_pause_${mode} PRIVATE -O2 -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
