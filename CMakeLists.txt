option(THUMBGEN_VERIFY_DEPENDENCY_VERSIONS "Require the versions used by the locked SDKs" OFF)
if(THUMBGEN_VERIFY_DEPENDENCY_VERSIONS)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/versions.json" THUMBGEN_DEPENDENCY_LOCK)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/versions.json")
    string(JSON THUMBGEN_EXPECTED_OIIO GET "${THUMBGEN_DEPENDENCY_LOCK}" OpenImageIO)
    string(JSON THUMBGEN_EXPECTED_JSON GET "${THUMBGEN_DEPENDENCY_LOCK}" nlohmann_json "${CMAKE_SYSTEM_NAME}")
    if(NOT "${OpenImageIO_VERSION}" STREQUAL "${THUMBGEN_EXPECTED_OIIO}" OR NOT "${nlohmann_json_VERSION}" STREQUAL "${THUMBGEN_EXPECTED_JSON}")
        message(FATAL_ERROR "Locked SDK versions required: OpenImageIO ${THUMBGEN_EXPECTED_OIIO}, nlohmann_json ${THUMBGEN_EXPECTED_JSON}; found ${OpenImageIO_VERSION}, ${nlohmann_json_VERSION}")
    endif()
endif()
set(THUMBGEN_SOURCE_REVISION "unknown")

find_package(Git QUIET)
if(Git_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.git")
    execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --short=12 HEAD
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE THUMBGEN_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND ${GIT_EXECUTABLE} status --porcelain --untracked-files=normal
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE THUMBGEN_GIT_CHANGES)
    if(THUMBGEN_GIT_CHANGES)
        set(THUMBGEN_SOURCE_DIRTY true)
    endif()
endif()
