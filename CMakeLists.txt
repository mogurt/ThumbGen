        file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${input}" THUMBGEN_DEEP_REFERENCE)
        # Separate literals avoid MSVC's per-literal size limit; concatenate only at test runtime.
        string(LENGTH "${THUMBGEN_DEEP_REFERENCE}" THUMBGEN_DEEP_REFERENCE_LENGTH)
        set(THUMBGEN_DEEP_REFERENCE_CHUNKS "")
        set(THUMBGEN_DEEP_REFERENCE_OFFSET 0)
        while(THUMBGEN_DEEP_REFERENCE_OFFSET LESS THUMBGEN_DEEP_REFERENCE_LENGTH)
            string(SUBSTRING "${THUMBGEN_DEEP_REFERENCE}" ${THUMBGEN_DEEP_REFERENCE_OFFSET} 4096 THUMBGEN_DEEP_REFERENCE_PART)
            string(APPEND THUMBGEN_DEEP_REFERENCE_CHUNKS "R\"DEEP_REFERENCE(${THUMBGEN_DEEP_REFERENCE_PART})DEEP_REFERENCE\",\n")
            math(EXPR THUMBGEN_DEEP_REFERENCE_OFFSET "${THUMBGEN_DEEP_REFERENCE_OFFSET}+4096")
        endwhile()
        set(THUMBGEN_DEEP_REFERENCE_SYMBOL "${symbol}")
        configure_file(tests/deep_reference.h.in "${output}" @ONLY)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
    endfunction()
